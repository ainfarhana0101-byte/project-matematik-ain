    <script>
        // Data generation arrays (Reactor tracking timeline from 0 to 20 hours)
        const timelineLabels = [];
        const sugarYieldData = [];
        const reactionRateData = [];

        // f(x): Hill Kinetics equation for reducing sugar accumulation
        function calculateYield(x) {
            return (50 * Math.pow(x, 2)) / (Math.pow(x, 2) + 25);
        }

        // f'(x): Analytical Derivative representing instantaneous conversion speed
        function calculateDerivative(x) {
            const numerator = 2500 * x;
            const denominator = Math.pow((Math.pow(x, 2) + 25), 2);
            return numerator / denominator;
        }

        // Fill data matrix for the chart at half-hour increments
        for (let hour = 0; hour <= 20; hour += 0.5) {
            timelineLabels.push(hour);
            sugarYieldData.push(calculateYield(hour).toFixed(2));
            reactionRateData.push(calculateDerivative(hour).toFixed(2));
        }

        // Chart.js Configuration Matrix
        const canvasCtx = document.getElementById('bioprocessChart').getContext('2d');
        const bioprocessChart = new Chart(canvasCtx, {
            type: 'line',
            data: {
                labels: timelineLabels,
                datasets: [
                    {
                        label: 'Sugar Yield f(x) (g/L)',
                        data: sugarYieldData,
                        borderColor: '#d97706', // Amber 600
                        backgroundColor: 'rgba(217, 119, 6, 0.08)',
                        fill: true,
                        tension: 0.2,
                        yAxisID: 'yAxisYield'
                    },
                    {
                        label: 'Hydrolysis Rate f\'(x) (g/L/hr)',
                        data: reactionRateData,
                        borderColor: '#2563eb', // Blue 600
                        borderDash: [6, 4],
                        fill: false,
                        tension: 0.2,
                        yAxisID: 'yAxisRate'
                    }
                ]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                animation: {
                    duration: 1800,
                    easing: 'easeOutCubic'
                },
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: 'Reaction Residence Time (Hours)',
                            font: { size: 11, weight: 'bold' }
                        }
                    },
                    yAxisYield: {
                        type: 'linear',
                        display: true,
                        position: 'left',
                        min: 0,
                        max: 55,
                        title: {
                            display: true,
                            text: 'Accumulated Reducing Sugars (g/L)',

                        }
                    }
                },
                plugins: {
                    legend: {
                        position: 'top',
                        labels: { boxWidth: 12, font: { size: 11 } }
                    }
                },
                // Click capture handler mapping coordinates back to data entries
                onClick: (event) => {
                    const activeElements = bioprocessChart.getElementsAtEventForMode(event, 'nearest', { intersect: false }, true);
                    if (activeElements.length > 0) {
                        const targetIndex = activeElements[0].index;
                        
                        const selectedHour = timelineLabels[targetIndex];
                        const selectedYield = parseFloat(sugarYieldData[targetIndex]);
                        const selectedRate = parseFloat(reactionRateData[targetIndex]);

                        evaluateBioprocessMetrics(selectedHour, selectedYield, selectedRate);
                    }
                }
            }
        });

        // Dynamic UI Dashboard Parsing Engine
        function evaluateBioprocessMetrics(hour, sugarYield, rate) {
            const panel = document.getElementById('dashboard-panel');
            const txtStage = document.getElementById('dash-stage');
            const badge = document.getElementById('dash-badge');
            
            const txtTime = document.getElementById('out-time');
            const txtYield = document.getElementById('out-yield');
            const txtRate = document.getElementById('out-rate');
            const txtExp = document.getElementById('out-explanation');

            // Set numeric values
            txtTime.innerHTML = `${hour.toFixed(1)} <span class="text-xs font-normal text-slate-400">hrs</span>`;
            txtYield.innerHTML = `${sugarYield.toFixed(2)} <span class="text-xs font-normal text-slate-400">g/L</span>`;
            txtRate.innerHTML = `${rate.toFixed(2)} <span class="text-xs font-normal text-slate-400">g/L/hr</span>`;

            // Reset dynamic border classes
            panel.className = "bg-slate-900 text-white p-6 rounded-2xl shadow-xl transition-all duration-300 border-l-8";

            // Phase logic evaluation (Lag, Log inflection, Diminishing Returns)
            if (hour <= 2.5) {
                panel.classList.add('border-blue-500');
                txtStage.textContent = "Process Status: Enzyme Structural Lag Phase";
                badge.className = "text-[10px] font-extrabold px-2.5 py-0.5 rounded-md bg-blue-950 text-blue-300";
                txtExp.innerHTML = `At hour <strong>${hour}</strong>, the process speed is slow at <strong>f'(x) = ${rate} g/L/hr</strong>. The enzymes are just starting to bind to the oil palm fibers. The derivative is small but climbing, meaning the reaction is picking up momentum.`;
            } 
            else if (hour > 2.5 && hour <= 6.5) {
                panel.classList.add('border-amber-500');
                txtStage.textContent = "Process Status: Maximum Kinetic Acceleration (Inflection Point)";
                badge.className = "text-[10px] font-extrabold px-2.5 py-0.5 rounded-md bg-amber-950 text-amber-300";
                txtExp.innerHTML = `At hour <strong>${hour}</strong>, you have hit the absolute sweet spot! The derivative peaks near here at <strong>f'(x) = ${rate} g/L/hr</strong>. This value is the slope's steepness peak. The enzyme system is cleaving biomass bonds at maximum velocity.`;
            } 
            else {
                panel.classList.add('border-rose-600');
                txtStage.textContent = "Process Status: Product Inhibition & Sluggish Yields";
                badge.className = "text-[10px] font-extrabold px-2.5 py-0.5 rounded-md bg-rose-950 text-rose-300";
                txtExp.innerHTML = `At hour <strong>${hour}</strong>, total sugar looks high (<strong>${sugarYield} g/L</strong>), but notice the derivative has dropped down to <strong>f'(x) = ${rate} g/L/hr</strong>. The reaction has severely slowed down because accumulated sugars are blocking the enzymes. Keeping the reaction running past this hour wastes factory time and utilities for tiny marginal returns.`;
            }
        }
